Installation 🥸

# 1. Enter the project
cd deepskysniffer

# 2. Create virtual environment
python3 -m venv venv

# 3. Activate it
source venv/bin/activate

# 4. Upgrade packaging tools
python3 -m pip install --upgrade pip setuptools wheel

# 5. Install your project from setup.py
pip install -e .

# 6. Run it
deepskysniffer
